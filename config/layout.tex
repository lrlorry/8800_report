% Shared thesis typography. These are house choices, not ANU regulations.
% Keep the Latin Modern text/math family supplied by TeX for portable builds.
\geometry{a4paper,left=27mm,right=27mm,top=28mm,bottom=28mm,
  headheight=18pt,headsep=7mm,footskip=12mm}
\setstretch{1.18}
\setlength{\parskip}{0pt plus 0.5pt}

\pagestyle{fancy}
\fancyhf{}
\fancyhead[L]{\footnotesize\color{black!65}\nouppercase{\leftmark}}
\fancyfoot[C]{\small\thepage}
\renewcommand{\headrulewidth}{0.2pt}
\renewcommand{\headrule}{\color{black!25}%
  \hrule width\headwidth height\headrulewidth\vskip-\headrulewidth}
\renewcommand{\chaptermark}[1]{\markboth{\thechapter\quad #1}{}}
\fancypagestyle{plain}{%
  \fancyhf{}\fancyfoot[C]{\small\thepage}
  \renewcommand{\headrulewidth}{0pt}}

\titleformat{\chapter}[display]
  {\normalfont\bfseries\raggedright}
  {\sffamily\small\thesischapterlabel}{9pt}{\fontsize{23}{28}\selectfont}
\titleformat{name=\chapter,numberless}[block]
  {\normalfont\bfseries\raggedright\fontsize{21}{26}\selectfont}{}{0pt}{}
\titlespacing*{\chapter}{0pt}{8pt}{23pt}
\titleformat{\section}
  {\normalfont\bfseries\raggedright\fontsize{14}{17}\selectfont}
  {\thesection}{0.7em}{}
\titlespacing*{\section}{0pt}{2.1ex plus .5ex minus .2ex}{0.9ex plus .2ex}
\titleformat{\subsection}
  {\normalfont\bfseries\raggedright\fontsize{12}{15}\selectfont}
  {\thesubsection}{0.7em}{}
\titlespacing*{\subsection}{0pt}{1.7ex plus .4ex minus .2ex}{0.7ex plus .2ex}
\titleformat{\subsubsection}
  {\normalfont\bfseries\raggedright}{\thesubsubsection}{0.7em}{}

\captionsetup{font={small,stretch=1.1},labelfont=bf,labelsep=period,
  justification=raggedright,singlelinecheck=false,skip=7pt}
\captionsetup[table]{position=top,skip=6pt}
\setlength{\textfloatsep}{16pt plus 3pt minus 3pt}
\setlength{\floatsep}{12pt plus 2pt minus 2pt}
\setlength{\intextsep}{14pt plus 3pt minus 3pt}
\setlength{\abovecaptionskip}{7pt}
\setlength{\belowcaptionskip}{0pt}
\setlist{itemsep=0.15em,topsep=0.35em,parsep=0pt}
\setlength{\bibsep}{0.4em}

% Compact navigational lists; full captions remain beside the figures/tables.
\makeatletter
\renewcommand*{\l@chapter}[2]{%
  \ifnum\c@tocdepth>\m@ne
    \addpenalty{-\@highpenalty}%
    \vskip 0.7em \@plus 0.1em
    \begingroup
      \parindent\z@ \rightskip\@pnumwidth \parfillskip-\@pnumwidth
      \leavevmode\bfseries
      \@tempdima 1.6em
      \advance\leftskip\@tempdima \hskip-\leftskip
      #1\nobreak\hfil\nobreak\hb@xt@\@pnumwidth{\hss #2}\par
      \penalty\@highpenalty
    \endgroup
  \fi}
\renewcommand*{\l@figure}{\@dottedtocline{1}{0em}{2.8em}}
\let\l@table\l@figure
\makeatother

% Preserve appendix terminology in automatic references as well as headings.
\AddToHook{cmd/appendix/after}{\crefalias{chapter}{appendix}}
